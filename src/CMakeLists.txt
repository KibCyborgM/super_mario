cmake_minimum_required(VERSION 3.16)
project(super_mario LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT DEFINED GAME_TYPE)
    if(WIN32)
        set(GAME_TYPE "WindowsConsole" CACHE STRING "Game backend" FORCE)
    else()
        set(GAME_TYPE "LinuxConsole" CACHE STRING "Game backend" FORCE)
    endif()
endif()
set_property(CACHE GAME_TYPE PROPERTY STRINGS WindowsConsole LinuxConsole QtGui)

if(GAME_TYPE STREQUAL "QtGui")
    set(CMAKE_AUTORCC ON)
    find_package(Qt6 REQUIRED COMPONENTS Widgets)
elseif(GAME_TYPE STREQUAL "WindowsConsole")
    if(NOT WIN32)
        message(WARNING "WindowsConsole builds only on Windows")
    endif()
elseif(GAME_TYPE STREQUAL "LinuxConsole")
    if(WIN32)
        message(WARNING "LinuxConsole builds only on Linux")
    endif()
else()
    message(FATAL_ERROR "Unknown GAME_TYPE: ${GAME_TYPE}")
endif()

if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
    add_compile_options(-finput-charset=utf-8 -fexec-charset=utf-8)
endif()

set(SOURCES
    main.cpp
)

if(GAME_TYPE STREQUAL "QtGui")
    qt_add_executable(${PROJECT_NAME} ${SOURCES})
else()
    add_executable(${PROJECT_NAME} ${SOURCES})
endif()

if(GAME_TYPE STREQUAL "WindowsConsole")
    target_compile_definitions(${PROJECT_NAME} PRIVATE WINDOWS_CONSOLE)
elseif(GAME_TYPE STREQUAL "LinuxConsole")
    target_compile_definitions(${PROJECT_NAME} PRIVATE LINUX_CONSOLE)
    target_link_libraries(${PROJECT_NAME} PRIVATE ncurses pthread)
elseif(GAME_TYPE STREQUAL "QtGui")
    target_compile_definitions(${PROJECT_NAME} PRIVATE QT_GUI)
    target_link_libraries(${PROJECT_NAME} PRIVATE Qt6::Widgets)
    if(WIN32)
        set_target_properties(${PROJECT_NAME} PROPERTIES WIN32_EXECUTABLE TRUE)
    endif()
endif()

add_subdirectory(os_controls)
add_subdirectory(controller)
add_subdirectory(levels)
add_subdirectory(model)
add_subdirectory(objects)
add_subdirectory(ui)
add_subdirectory(util)
